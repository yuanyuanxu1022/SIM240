def save(name,v):
    with (OUT/name).open('x') as f:json.dump(v,f,ensure_ascii=False,indent=2);f.write('\n')

protected=load(OLD/'protected_inputs_and_outputs.json')

records=[];configs=[]
for rec in b['prepared_runs']:
    run=Path(rec['output_directory']);cfg=evidence.kv(run/'run_config.txt');r=evidence.kv(run/'result.txt')
    configs.append({k:v for k,v in cfg.items() if k not in ['run_id','output_directory','direction','dx_nm']})
    if cfg['direction']!='T':continue
    p=evidence.exact_parameters(cfg)
    with (run/'diagnostics.csv').open() as f:rows=list(csv.DictReader(f))
    J=np.array([float(v['J']) for v in rows]);n=p['window'];window=J[-n:]
    K=p['nu']*J[-1]/p['a'];meanK=p['nu']*float(window.mean())/p['a']
    count=max(1,n//4);suffixes=[]
    for factor in [1,2,3]:
        if factor*n<=len(J):
            a=J[-factor*n:]
            suffixes.append(dict(windows=factor,steps=len(a),relative_span=float(np.ptp(a)/abs(a.mean())),
                                 first_to_final_relative=float(a[-1]/a[0]-1)))
    temporal=dict(run_id=rec['run_id'],dx_nm=float(cfg['dx_nm']),K_end_m3=K,K_last_window_mean_m3=meanK,
        terminal_std=float(window.std(ddof=1)/abs(window.mean())),terminal_span=float(np.ptp(window)/abs(window.mean())),
        final_vs_last_window_mean=float(J[-1]/window.mean()-1),
        last_quarter_mean_vs_previous_quarter=float(J[-count:].mean()/J[-2*count:-count].mean()-1),
        end_time_s=float(rows[-1]['time_s']),end_tstar=p['nu']*float(rows[-1]['time_s'])/p['h']**2,
        suffixes=suffixes,asymptotic_time_bias_bound_established=False)
    slice_data=np.genfromtxt(run/'field_slice.csv',delimiter=',',names=True)
    sliceK=p['nu']*math.fsum(float(v) for v in slice_data['ux'])*p['dx']**2/(p['P']*p['a'])
    boundary=evidence.check_boundary(cfg,evidence.kv(run/'boundary_audit.txt'))
    temporal.update(slice_K_m3=sliceK,slice_over_volume_K_minus_one=sliceK/K-1,
        independently_reenumerated_geometry=boundary,dt_s=p['dt'],H_m=p['H'],window_steps=p['window'],steps=len(rows)-1)
    records.append(temporal)

# Pure IEEE-754 repeated addition matching the A1 volume observation path,
# validated against all three historical native result values before extrapolation.
volumes=[]
template=evidence.kv(Path(b['prepared_runs'][-1]['run_config_path']))
for dxnm in ['5','2.5','1.25','0.625']:
    cfg=dict(template,dx_nm=dxnm);p=evidence.exact_parameters(cfg)
    dx=float(dxnm)*1e-9;term=(dx*dx)*dx
    seq=np.full(p['nodes'],term,dtype=np.float64)
    np.add.accumulate(seq,out=seq);total=float(seq[-1]);del seq
    nominal=float(template['period_nm'])*1e-9*(float(template['Ly_nm'])*1e-9)*(float(template['h_nm'])*1e-9)+(float(template['w_nm'])*1e-9)*(float(template['Ly_nm'])*1e-9)*(float(template['d_nm'])*1e-9)
    rel=total/nominal-1;entry=dict(dx_nm=float(dxnm),core_fluid_nodes=p['nodes'],cell_volume_term=term,
        repeated_binary64_volume=total,relative_to_native_nominal=rel,
        relative_to_exact_nominal=total/(p['P']*p['Ly']*p['H'])-1,
        absolute_relative_pass_1e_10=abs(rel)<=1e-10,dt_s=p['dt'],window_steps=p['window'],maximum_steps=p['cap'])
    if dxnm!='0.625':
        row=next(r for r in records if r['dx_nm']==float(dxnm))
        run=CASE/'runs'/row['run_id']/'steady';native=float(evidence.kv(run/'result.txt')['fluid_volume_m3'])
        entry['historical_native_exact_match']=native==total
        assert native==total,(dxnm,native,total)
    else:
        # One set of 19 binary64 population values per fluid node only.
        entry['one_population_set_fluid_only_bytes']=p['nodes']*19*8
        entry['native_execution_performed']=False
    volumes.append(entry)
