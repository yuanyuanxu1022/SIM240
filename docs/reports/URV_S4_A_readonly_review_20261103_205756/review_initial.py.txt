def guard(event,args):
    if event in ('subprocess.Popen','os.system','os.exec','os.posix_spawn','os.posix_spawnp'):
        raise AssertionError('No processes permitted in read-only S4 A review')
    if event=='open':
        path,mode,flags=args
        writes=(isinstance(mode,str) and any(x in mode for x in 'wxa+')) or (isinstance(flags,int) and flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND))
        if writes:
            assert isinstance(path,(str,bytes,os.PathLike)) and Path(path).resolve().is_relative_to(OUT),'non-report write denied'
    if event in ('os.remove','os.rmdir','os.mkdir','os.chmod','os.utime','os.rename'):
        paths=args[:2] if event=='os.rename' else args[:1]
        assert all(Path(p).resolve().is_relative_to(OUT) for p in paths),'non-report mutation denied'
sys.addaudithook(guard)
